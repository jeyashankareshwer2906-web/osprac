03

Experiment 3: Threads

Aim: To create and execute a thread.

Program
#include <stdio.h>
#include <pthread.h>

void *show(void *arg) {
    printf("Thread is running\n");
    return NULL;
}

int main(void) {
    pthread_t t;
    pthread_create(&t, NULL, show, NULL);
    pthread_join(t, NULL);
    printf("Main program ends\n");
    return 0;
}

Expected output / note: Thread is running Main program ends. Compile on Linux using: gcc file.c -pthread

Result: Thus, the experiment was studied and executed / simulated successfully.

Experiment 4: FCFS CPU Scheduling

Aim: To calculate waiting time and turnaround time using FCFS.

Program
#include <stdio.h>
int main(void) {
    int n, i, bt[20], wt[20], tat[20];
    printf("Enter number of processes: "); scanf("%d", &n);
    for (i = 0; i < n; i++) { printf("Burst time P%d: ", i+1); scanf("%d", &bt[i]); }
    wt[0] = 0;
    for (i = 1; i < n; i++) wt[i] = wt[i-1] + bt[i-1];
    printf("Process\tBT\tWT\tTAT\n");
    for (i = 0; i < n; i++) { tat[i] = wt[i] + bt[i]; printf("P%d\t%d\t%d\t%d\n", i+1, bt[i], wt[i], tat[i]); }
    return 0;
}

Expected output / note: The waiting time and turnaround time of each process are displayed.

Result: Thus, the experiment was studied and executed / simulated successfully.

Experiment 5: Priority Scheduling (Non-preemptive)

Aim: To schedule processes based on priority (smaller number means higher priority).

Program
#include <stdio.h>
int main(void) {
    int n, i, j, bt[20], p[20], id[20], wt[20], tat[20], t;
    printf("Enter number of processes: "); scanf("%d", &n);
    for (i=0;i<n;i++) { id[i]=i+1; printf("Burst time and priority for P%d: ",i+1); scanf("%d%d",&bt[i],&p[i]); }
    for (i=0;i<n-1;i++) for (j=i+1;j<n;j++) if (p[i]>p[j]) {
        t=p[i];p[i]=p[j];p[j]=t; t=bt[i];bt[i]=bt[j];bt[j]=t; t=id[i];id[i]=id[j];id[j]=t;
    }
    wt[0]=0; for(i=1;i<n;i++) wt[i]=wt[i-1]+bt[i-1];
    printf("Process\tBT\tPriority\tWT\tTAT\n");
    for(i=0;i<n;i++){tat[i]=wt[i]+bt[i];printf("P%d\t%d\t%d\t\t%d\t%d\n",id[i],bt[i],p[i],wt[i],tat[i]);}
    return 0;
}

Expected output / note: Processes are displayed in priority order with waiting and turnaround times.

Result: Thus, the experiment was studied and executed / simulated successfully.

Experiment 6: Round Robin Scheduling

Aim: To perform Round Robin CPU scheduling using a time quantum.

Program
#include <stdio.h>
int main(void) {
    int n, i, q, bt[20], rem[20], time=0, done;
    printf("Enter number of processes: "); scanf("%d", &n);
    for(i=0;i<n;i++){printf("Burst time P%d: ",i+1);scanf("%d",&bt[i]);rem[i]=bt[i];}
    printf("Enter time quantum: "); scanf("%d",&q);
    do {
        done=1;
        for(i=0;i<n;i++) if(rem[i]>0){done=0;if(rem[i]>q){time+=q;rem[i]-=q;}else{time+=rem[i];rem[i]=0;printf("P%d completed at time %d\n",i+1,time);}}
    } while(!done);
    return 0;
}

Expected output / note: Each process receives CPU time in turns and its completion time is displayed.

Result: Thus, the experiment was studied and executed / simulated successfully.
